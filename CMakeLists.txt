cmake_minimum_required(VERSION 3.20)
project(puzzle_fighter_native VERSION 0.1.0 LANGUAGES CXX)
set(PF_DECOMP_OUTPUT "${CMAKE_CURRENT_SOURCE_DIR}/output/decomp" CACHE PATH "Complete decomp output")
foreach(required "manifest.json" "source/include/puzzle_fighter" "data/generated/puzzle_fighter/generated/source_howto_sequence.inc")
  if(NOT EXISTS "${PF_DECOMP_OUTPUT}/${required}")
    message(FATAL_ERROR "Incomplete decomp output: ${required}. Run decomp first or use the AIO image.")
  endif()
endforeach()
find_package(SDL3 3.4.16 REQUIRED CONFIG)
add_library(puzzle_fighter_decomp INTERFACE)
target_compile_features(puzzle_fighter_decomp INTERFACE cxx_std_17)
target_include_directories(puzzle_fighter_decomp INTERFACE "${PF_DECOMP_OUTPUT}/source/include" "${PF_DECOMP_OUTPUT}/data/generated")
add_executable(puzzle-fighter-native src/main.cpp third_party/cps2recomp/sound/driver.cpp third_party/cps2recomp/sound/qsound.cpp)
target_compile_features(puzzle-fighter-native PRIVATE cxx_std_17)
target_include_directories(puzzle-fighter-native PRIVATE third_party)
target_link_libraries(puzzle-fighter-native PRIVATE puzzle_fighter_decomp SDL3::SDL3)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(puzzle-fighter-native PRIVATE -Wall -Wextra -Werror)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")
  # AppleClang flags the compact one-line if/else style as misleading
  # indentation; GCC doesn't. Linux builds are unaffected.
  target_compile_options(puzzle-fighter-native PRIVATE -Wno-misleading-indentation)
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/sdl-runtime.txt" CONTENT "$<TARGET_FILE:SDL3::SDL3>\n")
install(TARGETS puzzle-fighter-native RUNTIME DESTINATION .)
